6.1 CONFIGURACIÓN BASE
const API_BASE_URL = 'http://localhost:3000/api';
const TOKEN_KEY = 'bomberos_token';
const USER_KEY = 'bomberos_user';

const getToken = () => localStorage.getItem(TOKEN_KEY);

const getAuthHeaders = () => ({
  'Content-Type': 'application/json',
  'Authorization': `Bearer ${getToken()}`
});

6.2 SERVICIO DE AUTENTICACIÓN
class AuthService {
  static async login(email, password) {
    try {
      const response = await fetch(`${API_BASE_URL}/auth/login`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password })
      });
      
      const data = await response.json();
      
      if (data.success) {
        localStorage.setItem(TOKEN_KEY, data.data.token);
        localStorage.setItem(USER_KEY, JSON.stringify(data.data.user));
        return data.data;
      } else {
        throw new Error(data.message);
      }
    } catch (error) {
      throw error;
    }
  }

  static logout() {
    localStorage.removeItem(TOKEN_KEY);
    localStorage.removeItem(USER_KEY);
    window.location.href = '/login';
  }

  static isAuthenticated() {
    return !!getToken();
  }

  static getCurrentUser() {
    const user = localStorage.getItem(USER_KEY);
    return user ? JSON.parse(user) : null;
  }
}

6.3 SERVICIO DE FORMULARIOS
class FormulariosService {
  static async getMisFormularios(params = {}) {
    try {
      const queryString = new URLSearchParams(params).toString();
      const response = await fetch(
        `${API_BASE_URL}/formularios-necesidades/mis-formularios?${queryString}`,
        { headers: getAuthHeaders() }
      );
      
      const data = await response.json();
      return data.success ? data.data : null;
    } catch (error) {
      throw error;
    }
  }

  static async crearFormulario(formData) {
    try {
      const response = await fetch(`${API_BASE_URL}/formularios-necesidades`, {
        method: 'POST',
        headers: getAuthHeaders(),
        body: JSON.stringify(formData)
      });
      
      const data = await response.json();
      return data.success ? data.data : null;
    } catch (error) {
      throw error;
    }
  }

  static async getFormularioById(id) {
    try {
      const response = await fetch(
        `${API_BASE_URL}/formularios-necesidades/${id}`,
        { headers: getAuthHeaders() }
      );
      
      const data = await response.json();
      return data.success ? data.data : null;
    } catch (error) {
      throw error;
    }
  }
}

6.4 INTERCEPTOR PARA MANEJAR ERRORES
const handleApiResponse = async (response) => {
  if (response.status === 401) {
    AuthService.logout();
    return null;
  }
  
  if (!response.ok) {
    const errorData = await response.json();
    throw new Error(errorData.message || 'Error en la petición');
  }
  
  return response.json();
};

const apiCall = async (url, options = {}) => {
  try {
    const response = await fetch(url, options);
    return await handleApiResponse(response);
  } catch (error) {
    console.error('Error en API call:', error);
    throw error;
  }
};

6.5 EJEMPLO DE COMPONENTE DE LOGIN
const LoginComponent = () => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);

  const handleLogin = async (e) => {
    e.preventDefault();
    setLoading(true);
    
    try {
      const result = await AuthService.login(email, password);
      window.location.href = '/dashboard';
    } catch (error) {
      alert('Error en login: ' + error.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleLogin}>
      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        placeholder="Email"
        required
      />
      <input
        type="password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        placeholder="Contraseña"
        required
      />
      <button type="submit" disabled={loading}>
        {loading ? 'Iniciando...' : 'Iniciar Sesión'}
      </button>
    </form>
  );
};

6.6 EJEMPLO DE COMPONENTE DE LISTA
const FormulariosListComponent = () => {
  const [formularios, setFormularios] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const loadFormularios = async () => {
      try {
        const data = await FormulariosService.getMisFormularios();
        setFormularios(data.formularios);
      } catch (error) {
        console.error('Error cargando formularios:', error);
      } finally {
        setLoading(false);
      }
    };

    loadFormularios();
  }, []);

  if (loading) return <div>Cargando formularios...</div>;

  return (
    <div>
      {formularios.map(formulario => (
        <div key={formulario.id}>
          <h3>Formulario #{formulario.id}</h3>
          <p>Estado: {formulario.estado_nombre}</p>
          <p>Fecha: {new Date(formulario.fecha_creacion).toLocaleDateString()}</p>
        </div>
      ))}
    </div>
  );
};

================================================================================

9. CONSIDERACIONES IMPORTANTES
================================================================================

1. Manejo de Errores: Siempre implementa manejo de errores para tokens expirados
2. Loading States: Muestra estados de carga para mejor UX
3. Validación: Valida datos antes de enviar al servidor
4. Seguridad: Nunca almacenes información sensible en localStorage
5. Refresh Token: Considera implementar refresh tokens para sesiones largas
6. CORS: Asegúrate de que el backend permita peticiones desde tu frontend
7. HTTPS: En producción, usa siempre HTTPS
